# @group Shards and allocation

# @recipe Shards, problems first
# @tags unassigned initializing relocating red yellow
# UNASSIGNED shards come first, with the reason they lost their node.
GET _cat/shards?v&h=index,shard,prirep,state,unassigned.reason,unassigned.for,node&s=state:desc

# @recipe Why is a shard unassigned?
# @tags allocation explain red yellow
# Explains the first unassigned shard found (an error if there is none):
# which rule keeps it off each node.
GET _cluster/allocation/explain

# @recipe Why is this shard where it is?
# @tags allocation explain
# Same explanation for one shard in particular, assigned or not.
GET _cluster/allocation/explain
{
  "index": "${index}",
  "shard": 0,
  "primary": true
}

# @recipe Retry shards that gave up allocating
# @tags reroute failed max_retries
# After five failed attempts a shard stays unassigned until asked again: do
# it once the cause (disk, mapping, corrupted segment...) is fixed.
POST _cluster/reroute?retry_failed=true

# @recipe Disk usage and shard count per node
# @tags disk watermark balance
GET _cat/allocation?v&s=disk.percent:desc

# @recipe Disk watermarks in force
# @tags disk watermark flood read-only
# Above "low", no new shard goes to the node; above "high", shards are moved
# away from it; above "flood_stage", its indices become read-only.
GET _cluster/settings?include_defaults=true&filter_path=*.cluster.routing.allocation.disk*

# @recipe Recoveries in progress
# @tags recovery relocation initializing progress
GET _cat/recovery?v&active_only=true&h=index,shard,stage,type,source_node,target_node,bytes_percent,translog_ops_percent,time

# @recipe Relocating, initializing and unassigned shard counts
# @tags progress rebalance
GET _cluster/health?filter_path=status,relocating_shards,initializing_shards,unassigned_shards,delayed_unassigned_shards,active_shards_percent_as_number

# @recipe Largest shards
# @tags size store hot
GET _cat/shards?v&h=index,shard,prirep,store,docs,node&s=store:desc

# @recipe Stop shard allocation before restarting a node
# @tags rolling restart upgrade maintenance
# Keeps the cluster from rebuilding the node's replicas elsewhere while it
# is down. Re-enable it afterwards (next recipe).
PUT _cluster/settings
{
  "persistent": {
    "cluster.routing.allocation.enable": "primaries"
  }
}

# @recipe Re-enable shard allocation after a restart
# @tags rolling restart upgrade maintenance
PUT _cluster/settings
{
  "persistent": {
    "cluster.routing.allocation.enable": null
  }
}

# @recipe Move every shard off a node
# @tags drain exclude decommission
# The node's shards are relocated to the others; watch it empty out with
# "Disk usage and shard count per node". Cancel with the next recipe.
PUT _cluster/settings
{
  "persistent": {
    "cluster.routing.allocation.exclude._name": "${node}"
  }
}

# @recipe Cancel a node drain
# @tags exclude decommission
PUT _cluster/settings
{
  "persistent": {
    "cluster.routing.allocation.exclude._name": null
  }
}

# @recipe Segment replication lag
# @opensearch >=2.6
# @tags replica segrep
GET _cat/segment_replication?v
